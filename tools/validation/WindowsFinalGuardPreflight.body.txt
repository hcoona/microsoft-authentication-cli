# Fixed six-tool check; no caller-selected path, directory enumeration or load.
function Assert-FixedTool([string] $Path, [string] $Expected) {
    $item = Get-Item -LiteralPath $Path -Force
    if ($item.PSIsContainer) { exit 1 }
    $ancestor = $item
    while ($null -ne $ancestor) {
        if ($ancestor.Attributes -band [IO.FileAttributes]::ReparsePoint) { exit 1 }
        if ($ancestor.PSIsContainer) { $ancestor = $ancestor.Parent }
        else { $ancestor = $ancestor.Directory }
    }
    if ((Get-FileHash -LiteralPath $Path -Algorithm SHA256).Hash.ToLowerInvariant() -cne $Expected) { exit 1 }
}
